    <script>
        let currentStage = 0;
        const totalStages = 4;
        const sections = document.querySelectorAll('.stage-section');
        const indicators = document.querySelectorAll('.dot-indicator');

        function updateStages(index) {
            currentStage = Math.max(0, Math.min(index, totalStages - 1));
            
            sections.forEach((sec, idx) => {
                if (idx === currentStage) {
                    sec.classList.add('active');
                } else {
                    sec.classList.remove('active');
                }
            });

            indicators.forEach((dot, idx) => {
                if (idx === currentStage) {
                    dot.classList.add('bg-white', 'h-6', 'sm:h-8');
                    dot.classList.remove('bg-neutral-600', 'h-2.5', 'sm:h-3');
                } else {
                    dot.classList.remove('bg-white', 'h-6', 'sm:h-8');
                    dot.classList.add('bg-neutral-600', 'h-2.5', 'sm:h-3');
                }
            });

            sections[currentStage].scrollIntoView({ behavior: 'smooth' });
        }

        function goToStage(index) {
            updateStages(index);
        }

        // Обработка колесика мыши для десктопов
        let isScrolling = false;
        window.addEventListener('wheel', (e) => {
            // На мобильных устройствах wheel может срабатывать некорректно, исключаем узкие экраны по ширине
            if (window.innerWidth < 768) return;
            if (isScrolling) return;
            isScrolling = true;

            if (e.deltaY > 40) {
                if (currentStage < totalStages - 1) {
                    updateStages(currentStage + 1);
                }
            } else if (e.deltaY < -40) {
                if (currentStage > 0) {
                    updateStages(currentStage - 1);
                }
            }

            setTimeout(() => {
                isScrolling = false;
            }, 900);
        }, { passive: true });

        // Поддержка свайпов (touch) на мобильных телефонах
        let touchStartY = 0;
        window.addEventListener('touchstart', (e) => {
            touchStartY = e.touches[0].clientY;
        }, { passive: true });

        window.addEventListener('touchend', (e) => {
            let touchEndY = e.changedTouches[0].clientY;
            let diff = touchStartY - touchEndY;

            // Если свайп достаточно заметный
            if (Math.abs(diff) > 50) {
                if (diff > 0 && currentStage < totalStages - 1) {
                    updateStages(currentStage + 1);
                } else if (diff < 0 && currentStage > 0) {
                    updateStages(currentStage - 1);
                }
            }
        }, { passive: true });

        // Инициализация
        updateStages(0);
    </script>
